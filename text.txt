public class BondInfoArg {
    private String bondCode; // 債券代號 (必填)
}

public class BondInfoDto {
    private String bondCode;        // 債券代號
    private String bondShortName;   // 債券簡稱
    private LocalDate issueDate;    // 發行日
    private LocalDate maturityDate; // 到期日
    private BigDecimal couponRate;  // 票面利率
    private BigDecimal refYieldRate;// 參考殖利率
    private BigDecimal refPrice;    // 參考百元價
    private LocalDate evaluationDate;// 評價日期
}

public class BondInfoRs {
    private String returnCode;      // 狀態碼 (如: "0000" 代表成功)
    private String returnMessage;   // 訊息 (如: "查詢成功")
    private BondInfoDto data;       // 實際債券資料 (承載上述 DTO)
}
